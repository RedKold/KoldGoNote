```python
# 一个采用mcmc方法，Metropolis-Hastings的采样分布试探

def mcmc_tail_samples(
    mu, Sigma,
    VaR,
    n_samples=10000,
    burn_in=1000,
    proposal_scale=0.5,
    verbose=False
):
    """
    使用 Metropolis-Hastings 算法生成联合分布在 R <= -VaR 条件下的样本。
    """
    # 检查输入维度
    assert len(mu) == 2, "Only supports 2D case (R, r_i)"
    assert Sigma.shape == (2, 2), "Covariance matrix must be 2x2"

    # 定义目标分布
    target = multivariate_normal(mean=mu, cov=Sigma)

    # 初始化
    samples = []
    x = np.array([-VaR - np.random.exponential(), 0])  # 初始点：确保在尾部
    accepts = 0

    total_steps = n_samples + burn_in

    for i in range(total_steps):
        # 提案分布
        x_new = x + np.random.normal(scale=proposal_scale, size=2)

        # 约束：只接受 R <= -VaR 的样本
        if x_new[0] > -VaR:
            continue

        # 接受概率
        try:
            log_p_x = target.logpdf(x)
            log_p_x_new = target.logpdf(x_new)
            log_acc_ratio = log_p_x_new - log_p_x
            acc_ratio = np.exp(min(0, log_acc_ratio))  # 防止溢出
        except:
            continue  # 防止数值错误

        if np.random.rand() < acc_ratio:
            x = x_new
            if i >= burn_in:
                accepts += 1

        if i >= burn_in:
            samples.append(x.copy())

        if verbose and i % (total_steps // 10) == 0:
            print(f"Step {i}/{total_steps}, Acceptance rate so far: {accepts / max(1, i - burn_in):.3f}")

    samples = np.array(samples[-n_samples:])  # 确保返回 n_samples 个样本

    if verbose:
        print(f"Final acceptance rate: {accepts / n_samples:.3f}")

    return samples
```

```python
def compute_mcmc_mes(asset_returns, market_returns, alpha=0.05,
                     start_date=None, end_date=None,
                     n_samples=10000, burn_in=1000,
                     proposal_scale=0.01, plot_results=False):
    """
    使用 MCMC 方法计算 MES (Marginal Expected Shortfall)

    参数:
    - asset_returns: pandas.Series，资产收益率时间序列
    - market_returns: pandas.Series，市场收益率时间序列
    - alpha: float，分位数阈值 (默认0.05，表示最差5%)
    - start_date: str，开始日期 (格式: '2008-01-01')
    - end_date: str，结束日期 (格式: '2009-12-31')
    - n_samples: MCMC 采样数量
    - burn_in: 丢弃的前期样本数
    - proposal_scale: MCMC 提案分布尺度
    - plot_results: 是否绘制结果图

    返回:
    - mes: float，计算得到的MES值
    """

    # 1. 时间范围筛选
    if start_date:
        start_date_index = pd.to_datetime(start_date)
        if asset_returns.index.tz is not None:
            start_date_index = start_date_index.tz_localize(asset_returns.index.tz)
        asset_returns = asset_returns[asset_returns.index >= start_date_index]
        market_returns = market_returns[market_returns.index >= start_date_index]

    if end_date:
        end_date_index = pd.to_datetime(end_date)
        if asset_returns.index.tz is not None:
            end_date_index = end_date_index.tz_localize(asset_returns.index.tz)
        asset_returns = asset_returns[asset_returns.index <= end_date_index]
        market_returns = market_returns[market_returns.index <= end_date_index]

    # 2. 对齐时间索引，去除缺失值
    common_index = asset_returns.dropna().index.intersection(market_returns.dropna().index)
    asset_returns_aligned = asset_returns.loc[common_index]
    market_returns_aligned = market_returns.loc[common_index]

    print(f"样本数量: {len(common_index)}")
    print(f"时间范围: {common_index[0]} 到 {common_index[-1]}")

    # 3. 构造联合数据矩阵 [市场收益, 资产收益]
    data = np.column_stack([market_returns_aligned.values, asset_returns_aligned.values])

    # 4. 估计联合分布的参数
    mu = np.mean(data, axis=0)  # [市场收益均值, 资产收益均值]
    Sigma = np.cov(data, rowvar=False)  # 协方差矩阵

    print(f"估计的均值向量: {mu}")
    print(f"估计的协方差矩阵:\n{Sigma}")

    # 5. 计算 VaR (市场收益的 α 分位数)
    VaR = -np.percentile(market_returns_aligned, alpha * 100)
    print(f"计算得到的 VaR (α={alpha}): {VaR:.4f}")

    # 6. 使用 MCMC 生成尾部样本
    try:
        samples = distribution.mcmc_tail_samples(
            mu=mu,
            Sigma=Sigma,
            VaR=VaR,
            n_samples=n_samples,
            burn_in=burn_in,
            proposal_scale=proposal_scale,
            verbose=True
        )

        print(f"MCMC 采样完成，获得 {len(samples)} 个样本")

    except Exception as e:
        print(f"MCMC 采样失败: {e}")
        return None

    # 7. 计算 MES (尾部样本中资产收益的均值)
    if len(samples) > 0:
        mes = np.mean(samples[:, 1])  # samples[:, 1] 是资产收益
        print(f"MCMC-based MES: {mes:.6f}")

        # 显示统计信息
        print(f"资产收益标准差: {np.std(samples[:, 1]):.6f}")
        print(f"最小资产收益: {np.min(samples[:, 1]):.6f}")
        print(f"最大资产收益: {np.max(samples[:, 1]):.6f}")

        # 可视化结果（如果需要）
        if plot_results and len(samples) > 0:
            fig, axes = plt.subplots(1, 3, figsize=(15, 4))

            # 散点图
            axes[0].scatter(samples[:, 0], samples[:, 1], alpha=0.5, s=1)
            axes[0].axvline(-VaR, color='r', linestyle='--', label=f'VaR={-VaR:.3f}')
            axes[0].set_xlabel('Market Return (R)')
            axes[0].set_ylabel('Asset Return (r_i)')
            axes[0].set_title('MCMC Samples in Tail Region')
            axes[0].legend()

            # 资产收益分布
            axes[1].hist(samples[:, 1], bins=50, density=True, alpha=0.7, color='skyblue')
            axes[1].axvline(mes, color='r', linestyle='--', linewidth=2, label=f'MES={mes:.4f}')
            axes[1].set_xlabel('Asset Return')
            axes[1].set_ylabel('Density')
            axes[1].legend()
            axes[1].set_title('Asset Return Distribution in Tail')

            # 市场收益分布
            axes[2].hist(samples[:, 0], bins=50, density=True, alpha=0.7, color='lightgreen')
            axes[2].axvline(-VaR, color='r', linestyle='--', linewidth=2, label=f'VaR=-{-VaR:.3f}')
            axes[2].set_xlabel('Market Return')
            axes[2].set_ylabel('Density')
            axes[2].legend()
            axes[2].set_title('Market Return Distribution in Tail')

            plt.tight_layout()
            plt.show()

    else:
        print("警告: MCMC 未生成有效样本")
        mes = np.nan

    return mes
```
